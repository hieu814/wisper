cmake_minimum_required(VERSION 3.22)
project(flutter_whisper_cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/whisper.cpp" CACHE PATH "whisper.cpp source")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "Missing whisper.cpp. Run: dart run tool/setup.dart")
endif()
foreach(FEATURE AVX AVX2 FMA F16C BMI2 SSE42)
  set(GGML_${FEATURE} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WHISPER_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)
if(APPLE)
  add_library(flutter_whisper_cpp STATIC flutter_whisper_cpp.cpp)
else()
  add_library(flutter_whisper_cpp SHARED flutter_whisper_cpp.cpp)
endif()
target_link_libraries(flutter_whisper_cpp PRIVATE whisper)
target_include_directories(flutter_whisper_cpp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(flutter_whisper_cpp PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
  # 16 KB page compatibility, including when using older NDK linkers.
  target_link_options(flutter_whisper_cpp PRIVATE "-Wl,-z,max-page-size=16384")
endif()
